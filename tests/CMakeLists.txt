cmake_minimum_required(VERSION 3.16)
project(halow_stage1_native_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
option(SANITIZE "Address and Undefined Behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Werror -Wno-unused-parameter)
if(SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=address,undefined)
endif()
set(ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
include_directories(${ROOT}/components/nrc7292/include ${ROOT}/components/nrc7292/hspi ${ROOT}/components/nrc_video_packet/include ${ROOT}/components/hvc_clip/include ${ROOT}/sx_newah/L2BRIDGE_HSPI)
add_library(video ${ROOT}/components/nrc_video_packet/src/nrc_video_packet.c)
add_library(l2 ${ROOT}/components/nrc7292/src/nrc_l2_protocol.c)
add_library(clip ${ROOT}/components/hvc_clip/src/hvc_clip.c)
add_library(hspi ${ROOT}/components/nrc7292/hspi/nrc-hspi.c)
add_library(link_adapt ${ROOT}/esp32/main/link_adapt.c)
target_include_directories(link_adapt PUBLIC ${ROOT}/esp32/main)
target_link_libraries(link_adapt PUBLIC clip)
add_executable(test_link_adapt test_link_adapt.c)
target_link_libraries(test_link_adapt link_adapt)
add_test(NAME link_adapt COMMAND test_link_adapt)
target_compile_options(hspi PRIVATE -Wno-unused-function)
foreach(name packetizer h264)
 add_executable(test_${name} test_${name}.c)
 target_link_libraries(test_${name} video)
 add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(test_l2_protocol test_l2_protocol.c)
target_link_libraries(test_l2_protocol l2)
add_test(NAME l2_protocol COMMAND test_l2_protocol)
foreach(name hspi hspi_transport)
 add_executable(test_${name} test_${name}.c)
 target_link_libraries(test_${name} hspi)
 add_test(NAME ${name} COMMAND test_${name})
endforeach()
add_executable(test_clip test_clip.c)
target_link_libraries(test_clip clip video)
add_test(NAME clip COMMAND test_clip ${ROOT}/assets/test_h264.bin ${ROOT}/assets/test_mjpeg.bin ${ROOT}/assets/test_h264_1080p.bin)
add_executable(test_mac test_mac.c ${ROOT}/sx_newah/L2BRIDGE_HSPI/sx_mac.c)
add_test(NAME mac COMMAND test_mac)
add_executable(packet_fixture packet_fixture.c)
target_link_libraries(packet_fixture clip video)
foreach(name hspi_faults native_records)
 add_executable(test_${name} test_${name}.c)
 target_link_libraries(test_${name} hspi l2)
 add_test(NAME ${name} COMMAND test_${name})
endforeach()
find_package(Threads REQUIRED)
add_executable(test_esp_adapter test_esp_adapter.c ${ROOT}/components/nrc7292/src/nrc7292.c)
target_include_directories(test_esp_adapter PRIVATE esp_stubs)
target_compile_options(test_esp_adapter PRIVATE -Wno-misleading-indentation)
target_link_libraries(test_esp_adapter hspi l2 Threads::Threads)
add_test(NAME esp_adapter COMMAND test_esp_adapter)
set_tests_properties(esp_adapter PROPERTIES TIMEOUT 30)
add_executable(test_sx_endpoint test_sx_endpoint.c)
target_include_directories(test_sx_endpoint PRIVATE nrc_stubs)
target_compile_options(test_sx_endpoint PRIVATE -ffunction-sections -fdata-sections -Wno-unused-function)
target_link_options(test_sx_endpoint PRIVATE -Wl,--gc-sections)
target_link_libraries(test_sx_endpoint l2)
add_test(NAME sx_endpoint COMMAND test_sx_endpoint)
add_executable(test_esp_network test_esp_network.c)
target_include_directories(test_esp_network PRIVATE esp_stubs ${ROOT}/esp32/main)
target_compile_options(test_esp_network PRIVATE -ffunction-sections -fdata-sections)
target_link_options(test_esp_network PRIVATE -Wl,--gc-sections)
target_link_libraries(test_esp_network clip video link_adapt)
add_test(NAME esp_network COMMAND test_esp_network)
